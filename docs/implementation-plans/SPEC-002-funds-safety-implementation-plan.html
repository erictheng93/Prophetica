2. 4.3 帳本表結構（PostgreSQL）

2.1 表與欄位（實作 SQL）

-- 帳戶表（每個帳戶一列）
CREATE TABLE ledger_accounts (
    id              BIGSERIAL PRIMARY KEY,
    code            VARCHAR(128) NOT NULL UNIQUE,        -- 例: 'Player.Wallet.42'
    type            VARCHAR(32)  NOT NULL,               -- asset/liability/revenue/expense
    token_kind        VARCHAR(8)   NOT NULL DEFAULT 'COPPER', -- 銅幣內Token 類型
    parent_id       BIGINT       REFERENCES ledger_accounts(id),
    metadata        JSONB        NOT NULL DEFAULT '{}',
    created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    closed_at       TIMESTAMPTZ
);
CREATE INDEX idx_accounts_type    ON ledger_accounts(type);
CREATE INDEX idx_accounts_code    ON ledger_accounts(code);

-- 交易表（每個 business event 一列，以 idempotency_key 防重）
CREATE TABLE ledger_transactions (
    id                 BIGSERIAL PRIMARY KEY,
    txn_type           VARCHAR(64) NOT NULL,             -- deposit/order_lock/trade/settle/...
    idempotency_key    VARCHAR(128) NOT NULL UNIQUE,     -- 上游事件的唯一 ID
    correlation_id     VARCHAR(128),                     -- 關聯 trace
    description        TEXT,
    metadata           JSONB NOT NULL DEFAULT '{}',
    created_at         TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    posted_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_txn_type ON ledger_transactions(txn_type);
CREATE INDEX idx_txn_corr ON ledger_transactions(correlation_id);

-- 分錄表（每個交易至少兩列：借方 + 貸方）
CREATE TABLE ledger_entries (
    id              BIGSERIAL PRIMARY KEY,
    txn_id          BIGINT NOT NULL REFERENCES ledger_transactions(id),
    account_id      BIGINT NOT NULL REFERENCES ledger_accounts(id),
    direction       SMALLINT NOT NULL CHECK (direction IN (1,-1)),  -- 1=借方/+; -1=貸方/-
    amount          BIGINT  NOT NULL CHECK (amount > 0),             -- 內部單位（百分點）
    token_kind        VARCHAR(8) NOT NULL,
    balance_after   BIGINT NOT NULL,                                 -- 該帳戶累計餘額
    created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_entries_txn     ON ledger_entries(txn_id);
CREATE INDEX idx_entries_account ON ledger_entries(account_id, created_at);
注意：若實際落地有 schema 名稱變更，需同步更新 ADR-0003 / ADR-0006，以及 spec 內對照表與治理步驟。

3. Constraint 與 trigger（對應 §5 不變式）

3.1 INV-1：每筆交易借貸平衡

-- 不變式：每個 transaction 的所有 entries amount * direction 之和必為 0
CREATE OR REPLACE FUNCTION fn_ledger_balance_check()
RETURNS TRIGGER AS $$
DECLARE total BIGINT;
BEGIN
    SELECT COALESCE(SUM(amount * direction), 0) INTO total
    FROM ledger_entries
    WHERE txn_id = NEW.txn_id;
    IF total <> 0 THEN
        RAISE EXCEPTION 'Ledger imbalance for txn %: %', NEW.txn_id, total;
    END IF;
    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- DEFERRABLE 在 commit 時檢查，允許 transaction 內逐筆插入
CREATE CONSTRAINT TRIGGER trg_ledger_balance
AFTER INSERT ON ledger_entries
DEFERRABLE INITIALLY DEFERRED
FOR EACH ROW EXECUTE FUNCTION fn_ledger_balance_check();

4. Idempotency 實作

CREATE TABLE idempotency_records (
    key         VARCHAR(128) PRIMARY KEY,
    txn_id      BIGINT REFERENCES ledger_transactions(id),
    request_hash VARCHAR(64) NOT NULL,
    response    JSONB NOT NULL,
    status_code INT   NOT NULL,
    created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    expires_at  TIMESTAMPTZ NOT NULL
);
CREATE INDEX idx_idempo_expires ON idempotency_records(expires_at);
  • 保存期：7 天
  • 清理：每日 cron 清理過期 key（可用 job queue / cleanup worker）
  • Webhook 重放採同一 key（例如 gateway transaction_id）去重邏輯

4.1 MVP 手動銅幣調整與 audit_logs

  • POST /internal/ledger/manual-points-adjustments 建立 manual_points_adjustment，只接受 delta-only：增加 Y 銅幣或扣除 Y 銅幣，不提供設定絕對餘額。
  • admin / manager 可查詢、增加與扣除會員銅幣；affiliate 僅可增加會員銅幣，affiliate 查詢任意會員餘額或扣點必須拒絕。
  • 查詢操作寫 audit_logs.action = 'wallet_balance_read'；調整操作寫 audit_logs.action = 'manual_points_adjustment' 並將 audit_log_id 放入 ledger transaction metadata。
  • INV-6 對 Expense.ManualPointsGrant 的例外僅限 manual_points_adjustment 扣點、actor_role 為 admin / manager 且帶 audit_log_id；ADR-0006 v1.0.2 已同步。

5. 8.2 wallet 負餘額治理

-- 一般 wallet 餘額 check
ALTER TABLE ledger_entries
  ADD CONSTRAINT chk_wallet_nonneg
  CHECK (
    NOT (account_id IN (SELECT id FROM ledger_accounts WHERE type='liability' AND code LIKE 'Player.Wallet.%')
         AND balance_after < 0)
  ) NOT VALID;  -- 例外情境由 application 層特例放行

負餘額保留給 chargeback 例外，帳戶需立即進入風控流程與凍結。

6. 性能與一致性實作

6.1 寫入瓶頸（§9.1）

  • 撮合事件寫入可靠訊息佇列，Ledger Worker 批次消費。
  • 批次建議：100ms 一批、每批約 1000 events（依壓力測試調整）。
  • 時間分區策略：Phase 1 起直接以 ledger_entries.created_at 做 monthly range partition，透過 pg_partman 預建未來 3 個月 partition。
  • 12 個月以上 cold partition detach / archival 到 R2，再透過 foreign table 保留可查詢能力。
  • 每個 partition 內對 created_at 建 BRIN index；不得採「先建普通表，日後再 online partition migration」路徑。
  • 熱資料可複製到 read-replica 供查詢。

6.2 讀取與一致性路徑（§9.2 / §9.3）

  • 可使用 Redis 快取 wallet 最新餘額。
  • 寫入時同步 invalidate。
  • cache miss 回源 ledger，以最新 balance_after 還原。
  • 前端可結合 optimistic UI + WS push。
  • GET /api/v1/me/wallet?at_seq=N：後端等待指定撮合 seq 對應的 ledger event 消費後再回傳，供前端 reconciliation 使用。
  • GET /api/v1/me/wallet?consistent=true：客服、對帳與合規場景強制 strong consistency；後端需等待該玩家最近未消費 ledger event 全部 commit。
  • 若等待超過 SLA，API 回傳可重試錯誤並輸出 proph_ledger_event_lag_seconds span，不得直接回傳 stale balance 偽裝一致。
